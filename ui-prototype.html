📊 数据模型映射（SQLite 表结构）

-- 主密码与设置
CREATE TABLE settings (
  key TEXT PRIMARY KEY,
  value TEXT NOT NULL
);
-- key 值: master_password_hash, recovery_phrase, auto_lock_minutes, 
--         temp_expire_days, clipboard_clear_seconds, global_shortcut, lock_shortcut

-- 密码表（加密存储）
CREATE TABLE passwords (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  title TEXT NOT NULL,
  username TEXT NOT NULL,
  password TEXT NOT NULL,  -- AES-256 加密
  url TEXT,
  notes TEXT,              -- AES-256 加密
  created_at INTEGER NOT NULL,
  updated_at INTEGER NOT NULL
);

-- 软件表
CREATE TABLE apps (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  name TEXT NOT NULL,
  path TEXT NOT NULL,
  icon BLOB,               -- 图标二进制
  arguments TEXT,          -- 启动参数
  category_id INTEGER,
  custom_name TEXT,
  created_at INTEGER NOT NULL,
  FOREIGN KEY (category_id) REFERENCES app_categories(id)
);

-- 软件分区表
CREATE TABLE app_categories (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  name TEXT NOT NULL UNIQUE,
  icon TEXT,
  sort_order INTEGER DEFAULT 0
);

-- 文本片段表
CREATE TABLE snippets (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  title TEXT NOT NULL,
  content TEXT NOT NULL,
  language TEXT,           -- 代码语言（用于高亮）
  tags TEXT,
  created_at INTEGER NOT NULL,
  updated_at INTEGER NOT NULL
);

-- 临时内容表
CREATE TABLE temp_contents (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  title TEXT,
  content TEXT NOT NULL,
  expire_at INTEGER NOT NULL,  -- 过期时间戳
  created_at INTEGER NOT NULL
);

-- 置顶项表
CREATE TABLE pinned_items (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  type TEXT NOT NULL,      -- password/app/snippet/temp
  item_id INTEGER NOT NULL,
  sort_order INTEGER DEFAULT 0,
  UNIQUE(type, item_id)
);
